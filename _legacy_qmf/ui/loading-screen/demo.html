🎨 QMF Loading Screen Demo

Test loading screen với các cấu hình khác nhau

📖 Cách sử dụng

// Import module
import { QMFLoadingScreen } from 'https://ui.quizzman.com/qmf/ui/loading-screen/index.js';

// Auto mode - tự động chạy animation
QMFLoadingScreen.auto({
    siteName: 'My Site',
    subtitle: 'Tagline here',
    siteUrl: 'https://mysite.com',
    theme: 'dark'
});

// Manual mode - điều khiển thủ công
const loader = QMFLoadingScreen.create({ siteName: 'My Site' });
loader.show();
loader.setProgress(50, 'Loading data...');
loader.complete(); // Shows success then hides
